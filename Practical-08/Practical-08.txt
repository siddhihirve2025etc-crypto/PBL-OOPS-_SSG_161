#include <iostream>
#include <string>
using namespace std;

class Person
{
protected:
    string name;
    int age;

public:
    void getPersonDetails()
    {
        cout << "Enter name: ";
        cin >> name;

        cout << "Enter age: ";
        cin >> age;
    }

    void displayPersonDetails()
    {
        cout << "Name: " << name << "\n";
        cout << "Age: " << age << "\n";
    }
};

class Department
{
protected:
    string deptName;
    string role;

public:
    void getDeptDetails()
    {
        cout << "Enter department name: ";
        cin >> deptName;

        cout << "Enter job role: ";
        cin >> role;
    }

    void displayDeptDetails()
    {
        cout << "Department: " << deptName << "\n";
        cout << "Role: " << role << "\n";
    }
};

class Employee : public Person, public Department
{
private:
    int employeeId;
    double salary;

public:
    void getEmployeeDetails()
    {
        cout << "--- Enter Employee ID & Salary ---\n";

        cout << "Enter Employee ID: ";
        cin >> employeeId;

        cout << "Enter Salary: ";
        cin >> salary;

        cout << "\n--- Enter Personal Details ---\n";
        getPersonDetails();

        cout << "\n--- Enter Department Details ---\n";
        getDeptDetails();
    }

    void displayEmployeeProfile()
    {
        cout << "\n===================================\n";
        cout << "         EMPLOYEE PROFILE          \n";
        cout << "===================================\n";

        cout << "Employee ID: " << employeeId << "\n";

        displayPersonDetails();
        displayDeptDetails();

        cout << "Salary: " << salary << "\n";

        cout << "===================================\n";
    }
};

int main()
{
    Employee emp;

    emp.getEmployeeDetails();
    emp.displayEmployeeProfile();

    return 0;
}
